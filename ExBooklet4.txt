public int height(TreeNode node) {
     if (node == null) {
          return 0;
     } else {
          int lHeight = height(node.left);
          int rHeight = height(node.right);
          return Math.max(lHeight, rHeight) + 1;
     }
}

public int no_of_nodes(TreeNode node) {
     if (node == null) {
          return 0;
     } else {
          return 1 + no_of_nodes(node.left) + no_of_nodes(node.right);
     }
}

public boolean is_leaf(TreeNode node) {
     return (node.left == null && node.right == null;
}

public int no_of_leaves(TreeNode node) {
     if (node == null) {
          return 0;
     } else if (is_leaf(node)) {
          return 1;
     } else {
          return no_of_leaves(node.left) + no_of_leaves(node.right);
     }
}

public boolean is_full(TreeNode node) {
     if (node == null) {
          return true;
     }

     if (is_leaf(node)) {
          return true;
     }

     if (node.left != null && node.right != null) {
          return is_full(node.left) && is_full(node.right);
     }

     return false;

public boolean is_balanced(TreeNode node) {
     if (node == null) {
          return true;
     }

     int lHeight = height(node.left);
     int rHeight = height(node.right);

     if (Math.abs(lHeight - rHeight) <= 1 && is_balanced(node.left) && is_balanced(node.right)) {
          return true;
     }

     return false;

public BinaryTree<E> cloneAt(Node<E> current) {
     if (current == null) {
          return new BinaryTree<E>();
     }

     if (current.isLeaf()) {
          return new BinaryTree<E>(current.data);
     }

     return new BinaryTree<E>(current.data, cloneAt(current.left), cloneAt(current.right));

public ArrayList<ArrayList<Integer>> paths(Node<E> lr) {
     if (lr == null) {
          return new ArrayList<ArrayList<Integer>>();
     } else if (lr.isLeaf()) {
          ArrayList<ArrayList<Integer>> result = new ArrayList<ArrayList<Integer>>();
          result.add(new ArrayList<Integer>());

          return result;
     } else {
          ArrayList<ArrayList<Integer>> rResult = paths(lr.right);

          for (ArrayList<Integer> l : rResult) {
               l.add(0, 1);
          }

          ArrayList<ArrayList<Integer>> lResult = paths(lr.left);

          for (ArrayList<Integer> l : lResult) {
               l.add(0, 0);
          }

          lResult.addAll(rResult);
          return lResult;
     }
}

public void bfsHelper(E root) {
     if(root == null)
     return;

     Queue<Node> q =new LinkedList<Node>();

     q.add(root);
     trav.add(root);

     while(true) {
          int nodeCount = q.size();

          if (nodeCount == 0) {
               break;
          }
     }

     while(nodeCount > 0) {
          Node node = q.peek();
          trav.add(node);
          q.remove();

          if(node.left != null) {
               q.add(node.left);
          }

          if(node.right != null) {
               q.add(node.right);
               nodeCount--;
          }
     }
}

public ArrayList<E> bfs() {
     bfsHelper(root);
     return trav;
}

public ArrayList<ArrayList<E>> bfs2() {
     Queue<TreeNode<E>> queue = new LinkedList<>();
     ArrayList<ArrayList<E>> result = new ArrayList<>();

     if (root == null) {
          return result;
     }
     
     queue.offer(root);

     while (!queue.isEmpty()) {
          int levelSize = queue.size();
          ArrayList<E> currentLevel = new ArrayList<>();

          for (int i = 0; i < levelSize; i++) {
               TreeNode<E> currentNode = queue.poll();
               currentLevel.add(currentNode.value);

               if (currentNode.left != null) {
                    queue.offer(currentNode.left);
               }

               if (currentNode.right != null) {
                    queue.offer(currentNode.right);
               }
          }

          result.add(currentLevel);
     }

     return result;

     class TreeNode<E> {
          E value;
          TreeNode<E> left, right;

          TreeNode(E value) {
               this.value = value;
               left = null;
               right = null;
          }
     }
}

private ArrayList <BinaryTree<E>> projectLevel(int i, Node<E> current) {
     ArrayList<BinaryTree<E>> r = newArrayList<BTree<E>>();

     if (i == 0 && current != null) {
          r.add(cloneAt(current));
     } else {
          if (current != null) {
               r.addAll(projectLevel(i - 1, current.left));
               r.addAll(projectLevel(i - 1, current.right));
          }
     }

     return r ;
}
